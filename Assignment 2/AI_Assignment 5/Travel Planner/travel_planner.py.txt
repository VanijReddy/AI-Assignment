# ---------------------------------------------------------------------------
# Scoring function (enhanced with rating + accessibility)
# ---------------------------------------------------------------------------

def score_place(place, user):
    score = 0

    if place["category"] in user["interests"]:
        score += 10

    if place["cost"] <= user["max_place_cost"]:
        score += 4

    if place["time"] <= user["max_time_per_place"]:
        score += 3

    if place["area"] == user["preferred_area"]:
        score += 5

    # New: factor in rating (0-5 scaled to 0-5 points)
    score += place["rating"] - 3.5

    # New: accessibility filter
    if user.get("accessibility_required") and not place["accessible"]:
        score -= 20   # effectively removes inaccessible places

    return score


# ---------------------------------------------------------------------------
# Itinerary builder — groups places by area (reduces travel time)
# ---------------------------------------------------------------------------

def create_itinerary(user):
    scored = sorted(
        [(score_place(p, user), p) for p in PLACES_KB],
        key=lambda x: x[0],
        reverse=True
    )

    selected = []
    total_time = 0
    available = user["days"] * 8   # 8 hours per day

    for _, place in scored:
        if total_time + place["time"] <= available:
            selected.append(place)
            total_time += place["time"]

    # Group by area to reduce back-and-forth travel (new feature)
    from collections import defaultdict
    area_groups = defaultdict(list)
    for place in selected:
        area_groups[place["area"]].append(place)

    grouped = []
    for area, places in area_groups.items():
        grouped.extend(places)

    return grouped


# ---------------------------------------------------------------------------
# Day-wise schedule generator (new feature not in reference)
# ---------------------------------------------------------------------------

def build_day_schedule(places, days):
    schedule = {d: [] for d in range(1, days + 1)}
    hours_used = {d: 0 for d in range(1, days + 1)}
    daily_limit = 8

    for place in places:
        for day in range(1, days + 1):
            if hours_used[day] + place["time"] <= daily_limit:
                schedule[day].append(place)
                hours_used[day] += place["time"]
                break

    return schedule


# ---------------------------------------------------------------------------
# Food recommendation
# ---------------------------------------------------------------------------

def recommend_food(places):
    seen_areas = set()
    recommendations = []
    for place in places:
        if place["area"] not in seen_areas:
            seen_areas.add(place["area"])
            for food in FOOD_KB.get(place["area"], []):
                recommendations.append({
                    "food": food,
                    "drink": DRINK_KB.get(food, "Local Drink"),
                    "area": place["area"]
                })
    return recommendations


# ---------------------------------------------------------------------------
# Budget estimator
# ---------------------------------------------------------------------------

def estimate_budget(user, places):
    travel = DAILY_COST_KB[user["travel_style"]] * user["days"]
    attraction = sum(p["cost"] for p in places)
    food = 300 * user["days"]    # INR per day for food
    misc = 200 * user["days"]    # souvenirs, tips, etc.
    total = travel + attraction + food + misc
    return {
        "Transport": travel,
        "Attractions": attraction,
        "Food": food,
        "Miscellaneous": misc,
        "Total (INR)": total
    }


# ---------------------------------------------------------------------------
# Main plan generator
# ---------------------------------------------------------------------------

def generate_plan(user):
    places = create_itinerary(user)
    schedule = build_day_schedule(places, user["days"])
    foods = recommend_food(places)
    budget = estimate_budget(user, places)
    return {
        "city": "Hyderabad",
        "days": user["days"],
        "transport": TRANSPORT_KB[user["travel_style"]],
        "places": places,
        "schedule": schedule,
        "foods": foods,
        "budget": budget
    }


# ---------------------------------------------------------------------------
# Pretty printer
# ---------------------------------------------------------------------------

def print_plan(plan):
    print("\n" + "=" * 55)
    print("          AI TRAVEL PLAN — HYDERABAD")
    print("=" * 55)
    print(f"  City      : {plan['city']}")
    print(f"  Duration  : {plan['days']} day(s)")
    print(f"  Transport : {plan['transport']}")

    print("\n  DAY-WISE SCHEDULE")
    print("  " + "-" * 45)
    for day, places in plan["schedule"].items():
        print(f"  Day {day}:")
        if not places:
            print("    (rest / free time)")
        for p in places:
            cost_str = f"₹{p['cost']}" if p["cost"] > 0 else "Free"
            acc = "♿" if p["accessible"] else "  "
            print(f"    {acc} {p['name']:<30} {cost_str:>8}  {p['rating']}")

    print("\n  FOOD & DRINK RECOMMENDATIONS")
    print("  " + "-" * 45)
    for item in plan["foods"]:
        print(f"  [{item['area']}]  {item['food']}  →  {item['drink']}")

    print("\n  ESTIMATED BUDGET (INR)")
    print("  " + "-" * 45)
    for k, v in plan["budget"].items():
        print(f"  {k:<20}: ₹{v}")

    print("=" * 55)
